      import { test, expect } from "@playwright/test";
      import { selectFirstAvailableInEachGroup, expectAddToCartEnabled } from "./helpers/variants";

      const PRODUCT_SLUG = "isabella" as const; // LOCKED for the smoke
      const TEST_EMAIL = `e2e+${PRODUCT_SLUG}@test.calzoleriaprevenzano.it`;

      const FORM = {
        firstName: "E2E",
        lastName: "Test",
        address1: "Via Roma 1",
        city: "Roma",
        province: "RM",
        postalCode: "00100",
        phone: "+39 333 1234567",
      } as const;

      const CARD = {
        number: "4242 4242 4242 4242",
        exp: "12/34",
        cvc: "123",
        name: "E2E Test",
        zip: "00100",
      } as const;

      test("smoke: buy one Isabella sandalo end-to-end on live Railway", async ({ page }) => {
        // ── 1. Open product page ─────────────────────────────────────────────
        await page.goto(`/prodotti/${PRODUCT_SLUG}`);
        await expect(page.getByRole("heading", { name: /Isabella/i }).first()).toBeVisible();

        // ── 2. Pick variants (Tipo di Pelle = Laminato, then Colore = Verde acqua) ──
        await selectFirstAvailableInEachGroup(page);
        await expectAddToCartEnabled(page);

        // ── 3. Add to cart ──────────────────────────────────────────────────
        await page.getByRole("button", { name: /Aggiungi al carrello/i }).click();
        // Button text briefly becomes "Aggiunto!" — confirm we got past the click
        await expect(page.getByRole("button", { name: /Aggiunto|Aggiungi al carrello/i })).toBeVisible();

        // ── 4. Go to cart ───────────────────────────────────────────────────
        await page.goto("/carrello");
        await expect(page.getByText(/Isabella/i).first()).toBeVisible();

        // ── 5. Proceed to checkout ──────────────────────────────────────────
        await page.getByRole("link", { name: /Procedi al checkout/i }).click();
        await expect(page).toHaveURL(/\/checkout$/);

        // ── 6. Fill checkout form ───────────────────────────────────────────
        await page.getByLabel("Email").fill(TEST_EMAIL);
        await page.getByLabel("Nome").fill(FORM.firstName);
        await page.getByLabel("Cognome").fill(FORM.lastName);
        await page.getByLabel("Indirizzo", { exact: true }).fill(FORM.address1); // first "Indirizzo" only
        await page.getByLabel("Città").fill(FORM.city);
        await page.getByLabel("Provincia").fill(FORM.province);
        await page.getByLabel("CAP").fill(FORM.postalCode);
        // Phone is optional but harmless to fill
        const phone = page.getByLabel("Telefono");
        if (await phone.isVisible().catch(() => false)) await phone.fill(FORM.phone);

        // ── 7. Submit → wait for Stripe redirect ────────────────────────────
        // Submit button lives inside <OrderSummary>. The submit triggers POST /api/checkout
        // and then a 2 s setTimeout sets window.location.href to the Stripe URL.
        await Promise.all([
          page.waitForURL(/checkout\.stripe\.com/, { timeout: 60_000 }),
          page.getByRole("button", { name: /Paga|Vai al pagamento|Conferma|Procedi/i }).click(),
        ]);

        // ── 8. Fill Stripe Hosted Checkout (top-level page, NOT iframe) ─────
        // Locale is forced to en-US in playwright.config.ts so labels are stable.
        await page.getByLabel("Card number").fill(CARD.number);
        // Stripe sometimes uses one combined "Expiration" field, sometimes "MM / YY". Try both:
        const expField = page.getByLabel(/Expiration|MM \/ YY/);
        await expField.fill(CARD.exp);
        await page.getByLabel("CVC").fill(CARD.cvc);
        const nameField = page.getByLabel(/Cardholder name|Name on card/);
        if (await nameField.isVisible().catch(() => false)) await nameField.fill(CARD.name);
        const zipField = page.getByLabel(/ZIP|Postal code/);
        if (await zipField.isVisible().catch(() => false)) await zipField.fill(CARD.zip);

        // ── 9. Pay → wait for redirect back to /ordine-confermato ───────────
        await Promise.all([
          page.waitForURL(/\/ordine-confermato\?session_id=cs_test_/, { timeout: 90_000 }),
          page.getByTestId("hosted-payment-submit-button").click().catch(async () => {
            // fallback selector if Stripe drops the testid
            await page.getByRole("button", { name: /^Pay\b|Paga/i }).click();
          }),
        ]);

        // ── 10. Assert confirmation page renders order info ────────────────
        // Tolerate BOTH "Ordine confermato" and "Pagamento in elaborazione" — both are valid
        // success states for the smoke (webhook may not have fired yet).
        await expect(
          page.getByRole("heading", { name: /Ordine confermato|Pagamento in elaborazione/i }),
        ).toBeVisible({ timeout: 30_000 });

        // If we hit the "confermato" branch, an order number must be visible.
        const heading = await page
          .getByRole("heading", { name: /Ordine confermato|Pagamento in elaborazione/i })
          .textContent();
        if (heading?.match(/Ordine confermato/i)) {
          await expect(page.getByText(/Numero ordine:/i)).toBeVisible();
          // Order number is rendered after "Numero ordine:" — assert non-empty alphanumeric token.
          const orderLine = await page.getByText(/Numero ordine:/i).innerText();
          expect(orderLine).toMatch(/Numero ordine:\s*[A-Z0-9-]{4,}/);
          // Total > 0
          const totalText = await page.getByText(/Totale/i).innerText();
          expect(totalText).toMatch(/€\s*\d+(\.\d{2})?/);
        }
      });
