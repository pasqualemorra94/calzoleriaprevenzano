<objective>
Fix two production bugs surfaced by the Isabella smoke (IVA double-counted in totals; admin shows yellow "in attesa" instead of green "succeeded" after Stripe webhook), audit the customer-facing tax breakdown for consistency, then generalize the smoke into a parametric per-slug Playwright spec that discovers products at runtime and exercises the live Railway deploy end-to-end.

The admin label bug is a UX nuisance that makes succeeded payments look unprocessed. The parametric test surfaces the next class of bugs (variant-selection edge cases per product) so we can fix them in subsequent quick tasks.

Output: corrected `orders.server.ts`, corrected admin badge, corrected OrderSummary breakdown, new `all-products-purchase.spec.ts`, updated Playwright config + npm script, SUMMARY.md with diffs and run results.

    const BASE_URL = "https://calzoleria-prevenzano-production.up.railway.app";

    interface ProductListItem {
      slug: string;
      name: string;
    }

    interface ApiSuccess<T> {
      ok: true;
      data: T;
    }

    interface ProductsPage {
      items: ProductListItem[];
      total: number;
      page: number;

     */
    async function discoverSlugs(): Promise<string[]> {
      const ctx = await pwRequest.newContext({ baseURL: BASE_URL });
      const slugs: string[] = [];
      let page = 1;
      const perPage = 50;
      // Hard cap to avoid runaway loops.
      while (page <= 20) {
        const res = await ctx.get(`/api/products?page=${page}&perPage=${perPage}`);
        if (!res.ok()) break;
        const json = (await res.json()) as ApiSuccess<ProductsPage>;
        if (!json.ok) break;
        for (const item of json.data.items) {
          if (item.slug) slugs.push(item.slug);
        }
        if (page >= json.data.totalPages) break;
        page++;
      }
      await ctx.dispose();
      return slugs;
    }
